SYSCFG_DL_CAPTURE_0_init( ) 
 
    // tima0的时钟设置   （先忽略跳过）
    DL_TimerA_setClockConfig( CAPTURE_0_INST,    gCAPTURE_0ClockConfig );


    //tima0的捕获设置 
    //设置IFCTL_01[0]，不反相，捕获通道为CCP0 (水管通向外部引脚)
    //设置LOAD ， tima0的load值为65499
    //设置CCCTL_01[0] ，为capture， 捕获时不清0计数器， 每个 TIMCLK 走一步， 不用捕获边沿去重装计数器，上升沿捕获
    //CCP0设置为输入
    //设置 CTRCTL, 重载计数器听通道0的LCOND（不用捕获重装计数器），计数节奏听通道 0 的ACOND每个TIMCLK），
       计时器清0听通道0的ZCOND(不清0)，向下计数
       使能后从load开始计数，向下计数， 计数器数到0后自动重载并继续计数，停止计数器    

    DL_TimerA_initCaptureMode(CAPTURE_0_INST,    gCAPTURE_0CaptureConfig );


   //使能tima0的中断：向下触发门槛中断，zero中断
    //DL_TimerA_enableInterrupt(CAPTURE_0_INST , DL_TIMERA_INTERRUPT_CC0_DN_EVENT |  DL_TIMERA_INTERRUPT_ZERO_EVENT);     // 1<<4 | 1<<0 
        gptimer->CPU_INT.IMASK |= interruptMask;     //1<<4 | 1<<0   

    //使能tima0的时钟
    //DL_TimerA_enableClock(CAPTURE_0_INST);
    //     gptimer->COMMONREGS.CCLKCTL = (GPTIMER_CCLKCTL_CLKEN_ENABLED);
            CCLKCTL = 1<<0 ;      // CCLKCTL.0=1            //使能tima0的时钟

//设置CCPD，CCP0设置为输入

参数：TIMA0,      gCAPTURE_0CaptureConfig

DL_Timer_initCaptureMode(  *gptimer,   *config )
 
    Timer_Input_Chan_Config       captConfig;

    //取得DL_TIMER_INPUT_CHAN_0的一些初始化参数
    //DL_Timer_getInChanConfig(config->inputChan,   &captConfig);
    //   DL_Timer_getInChanConfig( DL_TIMER_INPUT_CHAN_0 ,   &captConfig );

            case DL_TIMER_INPUT_CHAN_0 ：  
               captConfig->index    = DL_TIMER_CC_0_INDEX;              //0

               captConfig->ccpInput = DL_TIMER_CC_IN_SEL_CCP0;          //0010<<0 

               captConfig->timerConfig =   (  GPTIMER_CTRCTL_CLC_CCCTL0_LCOND |                               // 000<<7
                                                                    GPTIMER_CTRCTL_CAC_CCCTL0_ACOND |        // 000<<10    
                                                                    GPTIMER_CTRCTL_CZC_CCCTL0_ZCOND |        // 000<<13
                                                                    GPTIMER_CTRCTL_CM_DOWN);                 // 00<<4

               captConfig->ccpInputDir = DL_TIMER_CC0_INPUT;                 // 0<<0


    //设置IFCTL_01[0]，不反相，捕获通道为CCP0 (水管通向外部引脚)
    //DL_Timer_setCaptureCompareInput( gptimer,   config->inputInvMode,  captConfig.ccpInput, captConfig.index  );
        TIMA0->COUNTERREGS.IFCTL_01[0] = 0<<7 |  0010<<0                     //bit.7=0,  bit.3-0 =2


    //设置LOAD ， tima0的load值为65499
    //DL_Timer_setLoadValue(gptimer, config->period);     //65499  
        gptimer->COUNTERREGS.LOAD =  65499;

    //设置CCCTL_01[0]，为capture，捕获时不清0计数器，计数节奏每个 TIMCLK 走一步，不用捕获边沿去重装计数器，上升沿捕获
    switch (config->captureMode)           //DL_TIMER_CAPTURE_MODE_EDGE_TIME  
   
        case DL_TIMER_CAPTURE_MODE_EDGE_TIME:
        case DL_TIMER_CAPTURE_MODE_EDGE_TIME_UP:

            // 设置CCCTL_01[0] ，为capture， 捕获时不清0计数器， 不用捕获边沿去重装计数器， 每个 TIMCLK 走一步，上升沿捕获
            //DL_Timer_setCaptureCompareCtl( gptimer,   DL_TIMER_CC_MODE_CAPTURE,                                                                     // 1<<17
                                             DL_TIMER_CC_ZCOND_NONE | DL_TIMER_CC_ACOND_TIMCLK |  DL_TIMER_CC_LCOND_NONE |  config->edgeCaptMode,     // 000<<12  |  000<<4  |  000<<8 |  001<<0 
                                                                       captConfig.index  );                                                           // DL_TIMER_CC_0_INDEX   //0

                 gptimer->COUNTERREGS.CCCTL_01[0]  =  1<<17 | 000<<12 | 000<<4 | 000<<8 | 1<<0;  //设置为capture，捕获时不清0计数器，计数节奏每个 TIMCLK 走一步，不用捕获边沿去重装计数器，上升沿捕获


        case DL_TIMER_CAPTURE_MODE_PERIOD_CAPTURE:
        case DL_TIMER_CAPTURE_MODE_PERIOD_CAPTURE_UP:
           。。。

        case DL_TIMER_CAPTURE_MODE_PULSE_WIDTH:
        case DL_TIMER_CAPTURE_MODE_PULSE_WIDTH_UP:
            。。。


    //设置CCPD，CCP0设置为输入
    //DL_Timer_setCCPDirection(gptimer, captConfig.ccpInputDir);       // DL_TIMER_CC0_INPUT   // 0<<0
           gptimer->COMMONREGS.CCPD =0<<0;                             //CCP0设置为输入



    switch (config->captureMode)            //DL_TIMER_CAPTURE_MODE_EDGE_TIME  
    
        case DL_TIMER_CAPTURE_MODE_EDGE_TIME_UP:
        case DL_TIMER_CAPTURE_MODE_PERIOD_CAPTURE_UP:
        case DL_TIMER_CAPTURE_MODE_PULSE_WIDTH_UP:
            captConfig.timerConfig =   (captConfig.timerConfig | GPTIMER_CTRCTL_CM_UP);   ///10<<4  (以后向上计数时有用)
            captConfig.timerConfig =   000<<7 | 000<<10 |000<<13 | 10<<4   //设置 CTRCTL, 重载计数器听通道0的LCOND（不用捕获重装计数器），计数节奏听通道 0 的ACOND每个TIMCLK），
                                                                             计时器清0听通道 0 的ZCOND(不清0)，向上计数

        default:   //yes  (本例)
   
            captConfig.timerConfig =   =   000<<7 | 000<<10  | 000<<13 |  00<<4    //设置CTRCTL, 重载计数器听通道0的LCOND(不用捕获重装计数器)，计数节奏听通道0的ACOND(每个TIMCLK)，
                                                                                     计时器清0听通道0的ZCOND(不清0)，向下计数
 
    //设置CTRCTL, 重载计数器听通道0的LCOND(不用捕获重装计数器)，计数节奏听通道0的ACOND(每TIMCLK)，计时器清0听通道0的ZCOND(不清0)，向下计数
       使能后从load开始计数，向下计数， 计数器数到0后自动重载并继续计数，停止计数器
    //DL_Common_updateReg(&gptimer->COUNTERREGS.CTRCTL,

        ( captConfig.timerConfig | GPTIMER_CTRCTL_CVAE_LDVAL |               //00<<28
            GPTIMER_CTRCTL_CM_DOWN | GPTIMER_CTRCTL_REPEAT_REPEAT_1 |        //00<<4 | 001 <<1
             config->startTimer),                                            //0<<0

        (GPTIMER_CTRCTL_CZC_MASK | GPTIMER_CTRCTL_CAC_MASK |
            GPTIMER_CTRCTL_CLC_MASK | GPTIMER_CTRCTL_CVAE_MASK |
            GPTIMER_CTRCTL_CM_MASK | GPTIMER_CTRCTL_REPEAT_MASK |
            GPTIMER_CTRCTL_EN_MASK));
 
        gptimer->COUNTERREGS.CTRCTL =   000<<7 |000<<10 |000<<13 | 00<<4 //设置CTRCTL, 重载计数器听通道0的LCOND(不用捕获重装计数器)，计数节奏听通道0的ACOND(每TIMCLK)，计时器清0听通道0的ZCOND(不清0)，向下计数
                                        | 00<<28 | 00<<4 | 001 <<1 | 0<<0       使能后从load开始计数，向下计数， 计数器数到0后自动重载并继续计数，停止计数器

CAPTURE_0_INST_IRQHandler()

    switch (DL_TimerG_getPendingInterrupt(CAPTURE_0_INST)) 
    
        case DL_TIMERG_IIDX_CC0_DN:
        
            if (gSynced == true)   //第3个是pwm的下降沿，   
            
                gCaptureCnt = DL_TimerG_getCaptureCompareValue(CAPTURE_0_INST, DL_TIMER_CC_0_INDEX);
                gCheckCaptures = true;
            
            
            else     //第1个是pwm的下降沿做同步信号     （第2个是pwm的上升沿，不会触发中断，但会拍照计数值到CC0）     
             
                gSynced = true;
            
            
            DL_TimerG_setTimerCount(CAPTURE_0_INST, gLoadValue);   // 收到同步信号后（或者完成一个pwm周期后），重新装载。 
            
            break;
            
            
        case DL_TIMERG_IIDX_ZERO:

            gSynced = false;    //没有检测到pwm信号
            break;

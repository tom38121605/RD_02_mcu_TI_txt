//设置IFCTL_01[0]，不反相，捕获通道为CCP0 (水管通向外部引脚)

//设置LOAD ，tima0的load值为2

//设置CCCTL_01[0] ，为compare， 捕获时不清0计数器， 计数节奏每个CCP的上升沿走一步，不用捕获边沿去重装计数器，关闭捕获

//设置CTRCTL, 重载计数器听通道0的LCOND(不用捕获重装计数器)，计数节奏听通道0的ACOND(每TIMCLK)，计时器清0听通道0的ZCOND(不清0)，向下计数
    使能后从load开始计数，向下计数， 计数器数到0后自动重载并继续计数，停止计数器




参数：TIMA0,      gCOMPARE_0CompareConfig

DL_Timer_initCompareMode(  gptimer,   config  )


    //DL_Timer_getInChanConfig( config->inputChan,  &inChanConfig );   //DL_TIMER_INPUT_CHAN_0

        case DL_TIMER_INPUT_CHAN_0:

            inChanConfig->index    = DL_TIMER_CC_0_INDEX;                // 0 ， CC0

            inChanConfig->ccpInput =  DL_TIMER_CC_IN_SEL_CCP0 ;          //0010<<0 

            inChanConfig->timerConfig =  ( GPTIMER_CTRCTL_CLC_CCCTL0_LCOND |      // 000<<7
                                     GPTIMER_CTRCTL_CAC_CCCTL0_ACOND |      // 000<<10 
                                     GPTIMER_CTRCTL_CZC_CCCTL0_ZCOND |      // 000<<13
                                     GPTIMER_CTRCTL_CM_DOWN);               // 00<<4

            inChanConfig->ccpInputDir = DL_TIMER_CC0_INPUT;                 // 0<<0



    //设置IFCTL_01[0]，不反相，捕获通道为CCP0 (水管通向外部引脚)
    //DL_Timer_setCaptureCompareInput( gptimer,  config->inputInvMode,  inChanConfig.ccpInput, inChanConfig.index );
        TIMA0->COUNTERREGS.IFCTL_01[0]  $= 0<<7 | 0010<<0                     //bit.7=0,  bit.3-0 =2


    //设置tima0的load值为2
    //DL_Timer_setLoadValue(gptimer, config->count);
       gptimer->COUNTERREGS.LOAD = 2;


     //设置CCCTL_01[0] ，为compare， 捕获时不清0计数器， 计数节奏每个CCP的上升沿走一步，不用捕获边沿去重装计数器，关闭捕获
    //DL_Timer_setCaptureCompareCtl( gptimer, DL_TIMER_CC_MODE_COMPARE,                                                                         // 0<<17
                                   DL_TIMER_CC_ZCOND_NONE | config->edgeDetectMode | DL_TIMER_CC_LCOND_NONE | DL_TIMER_CC_CCOND_NOCAPTURE ,   // 000<<12 | 001<<4 | 000<<8 | 000<<0
                                   inChanConfig.index );                                                                                      // DL_TIMER_CC_0_INDEX   //0

         gptimer->COUNTERREGS.CCCTL_01[0]  $=  0<<17 |  000<<12 | 001<<4 | 000<<8 | 000<<0;  //设置为compare， 捕获时不清0计数器， 计数节奏每个CCP的上升沿走一步，不用捕获边沿去重装计数器，关闭捕获



    //设置CTRCTL, 重载计数器听通道0的LCOND(不用捕获重装计数器)，计数节奏听通道0的ACOND(每TIMCLK)，计时器清0听通道0的ZCOND(不清0)，向下计数
       使能后从load开始计数，向下计数， 计数器数到0后自动重载并继续计数，停止计数器
    switch (config->compareMode)  //DL_TIMER_COMPARE_MODE_EDGE_COUNT
    
        case DL_TIMER_COMPARE_MODE_EDGE_COUNT:

            DL_Common_updateReg(&gptimer->COUNTERREGS.CTRCTL,
                   inChanConfig.timerConfig |
                    GPTIMER_CTRCTL_CVAE_LDVAL | GPTIMER_CTRCTL_CM_DOWN | GPTIMER_CTRCTL_REPEAT_REPEAT_1 |    //00<<28 | 00<<4 | 001 <<1
                    config->startTimer ,                                                                     //0<<0


                (GPTIMER_CTRCTL_CZC_MASK | GPTIMER_CTRCTL_CAC_MASK |
                    GPTIMER_CTRCTL_CLC_MASK | GPTIMER_CTRCTL_CVAE_MASK |
                    GPTIMER_CTRCTL_CM_MASK | GPTIMER_CTRCTL_REPEAT_MASK |
                    GPTIMER_CTRCTL_EN_MASK));
           
    
         gptimer->COUNTERREGS.CTRCTL $= 000<<7 |000<<10 |000<<13 | 00<<4  //设置CTRCTL, 重载计数器听通道0的LCOND(不捕获重装计数器)，计数节奏听通道0的ACOND(每TIMCLK)，计时器清0听通道0的ZCOND(不清0)，向下计数
                                        | 00<<28 | 00<<4 | 001 <<1 | 0<<0       使能后从load开始计数，向下计数， 计数器数到0后自动重载并继续计数，停止计数器

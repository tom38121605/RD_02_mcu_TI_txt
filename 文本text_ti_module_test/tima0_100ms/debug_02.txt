DL_Timer_initTimerMode( gptimer, config )

    //设置tima0的load值为 159
    //DL_Timer_setLoadValue(gptimer, config->period);
      gptimer->COUNTERREGS.LOAD = value;                    //159


    switch (config->timerMode) 

        case DL_TIMER_TIMER_MODE_PERIODIC:

             // 开启时从load计数
            //DL_Timer_setCounterValueAfterEnable(   gptimer, DL_TIMER_COUNT_AFTER_EN_LOAD_VAL );
              gptimer->COUNTERREGS.CTRCTL $= 00<<28;    // 开启时从load计数


    //设置tima0的cc0的初始门槛值为0
    //DL_Timer_setCaptureCompareValue(  gptimer, config->counterVal, DL_TIMER_CC_0_INDEX);
        gptimer->COUNTERREGS.CC_01[0]=0;

    // 设置为compare， 捕获时不清0计时器，不用捕获边沿去重装计数器，计数节奏每个 TIMCLK 走一步, 关闭捕获
    //DL_Timer_setCaptureCompareCtl(gptimer,  config->genIntermInt,   DL_TIMER_CC_ACOND_TIMCLK,  DL_TIMER_CC_0_INDEX);
        gptimer->COUNTERREGS.CCCTL_01[0] $=  0<<17 |  | 000<<12 |   | 000<<8 |  000<<4 |  000<<0;             // 设置为compare， 捕获时不清0计时器，不用捕获边沿去重装计数器，计数节奏每个 TIMCLK 走一步, 关闭捕获


    // 开启时从load计数，向下计数，向下数到 0 后自动重载， 停止定时器
   //gptimer->COUNTERREGS.CTRCTL =  config->timerMode | config->startTimer;
       gptimer->COUNTERREGS.CTRCTL $=   (00<<28) | 00 << 4 |  001<<1 | 0<<0;    // 开启时从load计数，向下计数，向下数到 0 后自动重载， 停止定时器

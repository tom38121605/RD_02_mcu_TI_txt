 main


   //外设初始化
   SYSCFG_DL_init().
  

   

   //---------------------PA0  event-------------------------------------------------------------------------------   

    //设置 SUB0CFG, 触发toggle， 引脚位PA0
    //DL_GPIO_configSubscriber(LED_PORT, DL_GPIO_SUBSCRIBER_INDEX_0, DL_GPIO_SUBSCRIBER_OUT_POLICY_TOGGLE, DL_GPIO_SUBSCRIBER0_PIN_0);  //0, 10<<8, 0000<16
       GPIOA->SUB0CFG  $= 10<<8 | 0000<<16;

    //把通道1连接到GPIOA： 通道1 （闹铃） --   PA0 toggle （开灯1）
    //DL_GPIO_setSubscriberChanID(LED_PORT, DL_GPIO_SUBSCRIBER_INDEX_0, GPIOA_EVENT_SUBSCRIBER_0_CHANNEL);   //  0 ,1   
      GPIOA->FSUB_0 = 1;         //把通道1连接到GPIOA

       //使能通道1的接收开关
    //DL_GPIO_enableSubscriber(LED_PORT, DL_GPIO_SUBSCRIBER_INDEX_0);  //0
      GPIOA->SUB0CFG |= 1<<0;



   
   //---------------------PA0  event----------end---------------------------------------------------------------------   
   
   
   
   //---------------------tima0  event-------------------------------------------------------------------------------   
   
   //设置RSTCTL，复位tima0外设 （如定时器计数值恢复到0）
   //DL_TimerA_reset(TIMER_0_INST);
     TIMA0->GPRCM.RSTCTL = (GPTIMER_RSTCTL_KEY_UNLOCK_W | GPTIMER_RSTCTL_RESETSTKYCLR_CLR | GPTIMER_RSTCTL_RESETASSERT_ASSERT);  //1b<<24 | 1<<1 | 1<<0

   //设置PWREN，使能tima0的power
   //DL_TimerA_enablePower(TIMER_0_INST);
     TIMA0->GPRCM.PWREN =  (GPTIMER_PWREN_KEY_UNLOCK_W | GPTIMER_PWREN_ENABLE_ENABLE);       //bit31-24 = 0x26  | 1<<0

   
    //设置tima0的时钟和分频
    //DL_TimerA_setClockConfig(TIMER_0_INST,  (DL_TimerA_ClockConfig *) &gTIMER_0ClockConfig);
        TIMA0->CLKSEL = 1<<3;
        TIMA0->CLKDIV = 001<<0;
        TIMA0->COMMONREGS.CPS = 255;
   

   
   //---DL_TimerA_initTimerMode(TIMER_0_INST,  (DL_TimerA_TimerConfig *) &gTIMER_0TimerConfig);---
   
   
      //设置LOAD，tima0的load值为 159
      //DL_Timer_setLoadValue(gptimer, config->period);
        TIMA0->COUNTERREGS.LOAD = 62499;                    
         
   
      // 设置CTRCTL，开启时从load计数
      //DL_Timer_setCounterValueAfterEnable(   gptimer, DL_TIMER_COUNT_AFTER_EN_LOAD_VAL );
        TIMA0->COUNTERREGS.CTRCTL &= ~(3<<28);    
        TIMA0->COUNTERREGS.CTRCTL |= 00<<28;    


      //设置CC_01，tima0的cc0的初始门槛值为0
      //DL_Timer_setCaptureCompareValue(  gptimer, config->counterVal, DL_TIMER_CC_0_INDEX);
        TIMA0->COUNTERREGS.CC_01[0]=0;
        

      // 设置CCCTL_01，为compare， 捕获时计数值不置0，捕获时计数值不置load，计数节奏每个 TIMCLK 走一步, 关闭捕获
      //DL_Timer_setCaptureCompareCtl(gptimer,  config->genIntermInt,   DL_TIMER_CC_ACOND_TIMCLK,  DL_TIMER_CC_0_INDEX);
        TIMA0->COUNTERREGS.CCCTL_01[0] &=  ~(1<<17 | 7<<12 | 7<<8 | 7<<4 | 7<<0);  
        TIMA0->COUNTERREGS.CCCTL_01[0] |=  0<<17 | 000<<12 | 000<<8 | 000<<4 | 000<<0;  
        
       
      // 设置CTRCTL，开启时从load计数，向下计数，向下数到 0 后时钟继续走， 停止定时器
      //gptimer->COUNTERREGS.CTRCTL =  config->timerMode | config->startTimer;
       TIMA0->COUNTERREGS.CTRCTL &=   ~(3<<28 | 3 << 4 |  7<<1 | 1<<0);  
       TIMA0->COUNTERREGS.CTRCTL |=   00<<28 | 00 << 4 |  001<<1 | 0<<0;  
       


    //使能tima0的中断：zero中断
    //DL_TimerA_enableInterrupt(TIMER_0_INST , DL_TIMERA_INTERRUPT_ZERO_EVENT);
        TIMA0->CPU_INT.IMASK |= 1<<0; 

    //使能tima0的时钟
    //DL_TimerA_enableClock(TIMER_0_INST);
       TIMA0->COMMONREGS.CCLKCTL =  1<<0;            //使能tima0的时钟

   
    // 触发事件tima0 zero（6点）  -- 广播站1   （把苹果装入箩筐）   
    //DL_TimerA_enableEvent(TIMER_0_INST, DL_TIMERA_EVENT_ROUTE_1, (DL_TIMERA_EVENT_ZERO_EVENT));
      TIMA0->GEN_EVENT0.IMASK  |= 1<<0;         //把苹果装入箩：tima0 zero（6点）  -- 广播站1
      
      
    //把事件zero连接到通道1： 触发事件tima0 zero（6点）  -- 广播站1  --   通道1 （闹铃）
    //DL_TimerA_setPublisherChanID(TIMER_0_INST, DL_TIMERA_PUBLISHER_INDEX_0, TIMER_0_INST_PUB_0_CH);
      TIMA0->FPUB_0 =1;        //把事件zero连接到通道1
      
    
    
   //---------------------tima0  event----------------end---------------------------------------------------------------


   
   NVIC_SetPriority(TIMER_0_INST_INT_IRQN, 1);
   //NVIC_EnableIRQ(TIMER_0_INST_INT_IRQN);


   //设置CTRCTL，开启定时器tima0
   //DL_TimerA_startCounter(TIMER_0_INST);
     TIMA0->COUNTERREGS.CTRCTL |= 1<<0;
